
        # 7. V3 Scorer — re-score gate-approved candidates.
        # Brock 2026-05-18: SURGICAL demotion. The V3 scorer can still
        # veto candidates that DON'T map to a live trade-tier cell
        # (legacy/observe path). For routing-approved candidates (cell
        # in v10_fast* at Seed-Pro+ tier), the scorer is demoted to
        # ranker-only. BLOCKED tier (physical hard-blocks like spread)
        # always drops regardless of routing approval.
        TRADE_TIERS = {"Seed-Pro", "Pro", "Elite"}

        if self.scorer is not None:
            scored: list[TradeCandidate] = []
            for candidate in approved:
                result = self.scorer.score(ctx, candidate)
                routing_cell_tier = candidate.metadata.get("routing_cell_tier")
                routing_approved = routing_cell_tier in TRADE_TIERS

                # SCORER VETO REMOVED 2026-05-27 (Brock'"'"'s call):
                # The scorer is now PURELY a ranker. BLOCKED + REJECT tiers
                # no longer drop candidates. The score still gets attached
                # to the candidate so the best-by-score selection at the
                # end of this loop still works. Log full context so we can
                # see what the old scorer would have killed.
                if result.tier in ("BLOCKED", "REJECT"):
                    log.info(
                        "[V3_SCORER] scorer_bypassed_brock_2026_05_27 "
                        "symbol=%s strategy=%s tier=%s raw_score=%d "
                        "(would_have_blocked_at_old_floor)",
                        symbol, candidate.strategy, result.tier, result.score,
                    )
                    candidate.metadata["v3_bypassed_by_brock_override"] = True
                    candidate.metadata["v3_would_have_rejected_at"] = result.score
                # Replace strategy's fixed score with V3 score
                candidate.score = float(result.score)
                candidate.metadata["v3_tier"] = result.tier
                candidate.metadata["v3_breakdown"] = result.breakdown
                candidate.metadata["v3_pairs"] = result.pairs_total

                # ML strategy filter (post-scorer). Predicts P(win) for
                # opt-in strategies. Mode=observe logs only; mode=hard
                # actually drops candidates below threshold.
                if self.ml_filter is not None:
                    ml = self.ml_filter.evaluate(symbol, candidate, ctx, result)
                    candidate.metadata["ml_filter"] = ml
                    if ml.get("vetoed"):
                        self._emit(
                            CANDIDATE_REJECTED,
                            symbol=symbol,
                            strategy=candidate.strategy,
                            reason=f"ML_FILTER:{ml.get('reason','low_p')}",
                        )
                        continue
                scored.append(candidate)

                # DBB candidate audit — log every dbb_trend candidate for calibration
                if candidate.strategy == "dbb_trend":
                    self._emit(
                        "DBBCandidateScored",
                        symbol=symbol,
                        strategy=candidate.strategy,
                        direction=candidate.direction.value if hasattr(candidate.direction, 'value') else str(candidate.direction),
                        entry_type=candidate.metadata.get("entry_type", ""),
                        bb_state=candidate.metadata.get("bb_state", ""),
                        base_score=result.base_score,
                        final_score=result.score,
                        tier=result.tier,
                        breakout_level=candidate.metadata.get("breakout_level", 0),
                        pullback_depth_pct=candidate.metadata.get("pullback_depth_pct", 0),
                        rr=round(candidate.risk_reward, 2),
                        management_profile=candidate.metadata.get("management_profile", ""),
                    )
            approved = scored

        if not approved:
            return None
